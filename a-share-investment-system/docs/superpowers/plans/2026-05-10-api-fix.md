@router.post("/run")
def run_analysis(data: dict):
    """创建分析任务（异步后台执行）"""
    code = data.get("stock_code", "").strip()
    if not code:
        return {"ok": False, "error": "stock_code required"}
    task_id = str(uuid.uuid4())[:8]
    try:
        from models import AnalysisTask, get_session
        session = get_session()
        task = AnalysisTask(task_id=task_id, stock_code=code,
                            stock_name=data.get("stock_name", code),
                            status="running", progress=0)
        session.add(task)
        session.commit()
        session.close()

        # 后台执行分析
        def _run():
            try:
                res = analyze_stock(code)
                s2 = get_session()
                t = s2.query(AnalysisTask).filter_by(task_id=task_id).first()
                if t:
                    t.status = "completed"
                    t.progress = 100
                    t.signal = res.get("signal", "")
                    t.confidence = res.get("confidence", 0)
                    t.result_json = json.dumps(res, ensure_ascii=False, default=str)
                    t.finish_time = datetime.now()
                    s2.commit()
                s2.close()
            except Exception as e:
                s2 = get_session()
                t = s2.query(AnalysisTask).filter_by(task_id=task_id).first()
                if t:
                    t.status = "failed"
                    t.error = str(e)
                    s2.commit()
                s2.close()

        threading.Thread(target=_run, daemon=True).start()
        return {"ok": True, "task_id": task_id, "status": "running"}
    except Exception as e:
        return {"ok": False, "error": str(e)}

    code = data.get("stock_code", "").strip()
    if not code:
        return {"ok": False, "error": "stock_code required"}
    try:
        from models import Watchlist, get_session
        session = get_session()
        existing = session.query(Watchlist).filter_by(stock_code=code).first()
        if existing:
            session.close()
            return {"ok": False, "error": "already exists"}
        item = Watchlist(stock_code=code, stock_name=data.get("stock_name", code),
                         created_at=datetime.now())
        session.add(item)
        session.commit()
        session.close()
        return {"ok": True}
    except Exception as e:
        return {"ok": False, "error": str(e)}
